    setConstantMonth(id: string, index: number, raw: string): void {
        const current = this.constants().find((item) => item.id === id);
        if (!current || index < 0 || index >= 12) {
            return;
        }
        const months = [...current.months];
        months[index] = raw ?? '';
        this.patchConstant(id, {months});
    }

    /**
     * Amount for a constant in a month. Markers `↑` / `=↑` / `=vorzeile` walk
     * to the previous month (explicit API; formerly YearComponent monkey-patch).
     */
    constantAmount(item: ConstantDef, monthTitle: string): number {
        const months = CONSTANT_MONTHS;
        let index = months.indexOf(monthTitle as typeof months[number]);
        const seen = new Set<number>();
        while (index >= 0) {
            if (seen.has(index)) {
                return 0;
            }
            seen.add(index);
            const raw = item.months[index] ?? '';
            if (this.isPrevMonthMarker(raw)) {
                index -= 1;
                continue;
            }
            return this.evaluateAmount(raw, months[index]);
        }
        return 0;
    }
